```javascript
function longTask() {
    return new Promise((resolve) => {
        setTimeout(() => {
            resolve("任务完成");
        }, 10000); // 10秒后完成任务
    });
}

async function handleUserInput(input) {
    if (input === "start") {
        const result = await startLongTask();
        console.log(result);
    } else {
        console.log("继续对话");
    }
}
```
